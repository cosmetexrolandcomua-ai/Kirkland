  function showToast(msg) {
    const toast = document.getElementById('bankapp-toast');
    if (!toast) return;
    toast.textContent = msg;
    toast.classList.add('is-visible');
    clearTimeout(toast._timer);
    toast._timer = setTimeout(function () { toast.classList.remove('is-visible'); }, 1800);
  }

  function copyText(text) {
    if (navigator.clipboard && navigator.clipboard.writeText) {
      return navigator.clipboard.writeText(text);
    }
    return new Promise(function (resolve, reject) {
      try {
        const ta = document.createElement('textarea');
        ta.value = text;
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        document.execCommand('copy');
        document.body.removeChild(ta);
        resolve();
      } catch (e) { reject(e); }
    });
  }

  function attachHandlers(amount, reference) {
    document.querySelectorAll('.bankapp-req-row').forEach(function (row) {
      row.addEventListener('click', function () {
        const target = row.getAttribute('data-copy-target');
        const valueEl = document.querySelector('[data-copy="' + target + '"]');
        if (!valueEl) return;
        copyText(valueEl.textContent.trim())
          .then(function () {
            showToast('Скопійовано!');
            row.classList.add('is-copied');
            clearTimeout(row._copyTimer);
            row._copyTimer = setTimeout(function () {
              row.classList.remove('is-copied');
            }, 1500);
          })
          .catch(function () { showToast('Не вдалось скопіювати'); });
      });
    });

    const copyAllBtn = document.querySelector('.bankapp-copy-all');
    if (copyAllBtn) {
      copyAllBtn.addEventListener('click', function () {
        const text =
          'Отримувач: ' + CONFIG.beneficiary + '\n' +
          'ЄДРПОУ: ' + CONFIG.edrpou + '\n' +
          'IBAN: ' + CONFIG.iban + '\n' +
          'Сума: ' + amount.toFixed(2) + ' грн\n' +
          'Призначення: ' + reference;
        copyText(text)
          .then(function () { showToast('Усі реквізити скопійовано'); })
          .catch(function () { showToast('Не вдалось скопіювати'); });
      });
    }

    const printBtn = document.getElementById('bankapp-print');
    if (printBtn) {
      printBtn.addEventListener('click', function () { window.print(); });
    }
  }

  function initBankapp() {
    if (!isTargetPage()) {
      log('Не на success-сторінці. Path:', window.location.pathname);
      return;
    }
    if (!isPaymentMethodMatch()) {
      log('Метод оплати не знайдено');
      return;
    }
    const orderId = extractOrderId();
    if (!orderId) {
      log('Номер замовлення не знайдено');
      return;
    }
    const amount = extractOrderAmount();
    if (!amount) {
      log('Сума не знайдена');
      return;
    }
    if (document.getElementById('bankapp-widget')) {
      log('Віджет уже вставлений');
      return;
    }

    const reference = CONFIG.referenceTemplate.replace('{order_id}', '№' + orderId);

    log('Замовлення №' + orderId + ', сума ' + amount + ' грн');

    addBankappStyles();

    const wrapper = document.createElement('div');
    wrapper.innerHTML = buildHtml(orderId, amount, reference);
    const widget = wrapper.firstChild;

    const targets = [
      document.querySelector('main .wrapper'),
      document.querySelector('main'),
      document.querySelector('#page'),
      document.querySelector('.page'),
      document.body
    ];
    for (let i = 0; i < targets.length; i++) {
      if (targets[i]) {
        targets[i].appendChild(widget);
        log('Віджет вставлено');
        break;
      }
    }

    attachHandlers(amount, reference);
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', function () { setTimeout(initBankapp, 200); });
  } else {
    setTimeout(initBankapp, 200);
  }
})();
